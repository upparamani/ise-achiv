Google Drive Proofs & Sheets Sync

        
      

      

        
Connect a Google Apps Script Web App to automatically save scanned certificates into your Google Drive folder and store the Drive file link in each Admin record.

        

          
          
        

        

          
          
        

        

          View / Copy Apps Script (Code.gs) to Save Scanned Docs to Drive
          

            
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var folderId = data.driveFolderId || '1MkZGPXGn_37MOyoMzsU8EteWxrcNHEYH';
  var driveUrl = data.driveUrl || '';
  if (data.proofDataUrl && data.proofDataUrl.indexOf('base64,') > -1) {
    var parts = data.proofDataUrl.split('base64,');
    var mime = parts[0].split(':')[1].split(';')[0];
    var bytes = Utilities.base64Decode(parts[1]);
    var safeName = (data.name || 'Certificate').replace(/[^a-zA-Z0-9_-]/g, '_');
    var blob = Utilities.newBlob(bytes, mime, safeName + '_' + Date.now() + '.png');
    var file = DriveApp.getFolderById(folderId).createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    driveUrl = file.getUrl();
  }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  sheet.appendRow([
    new Date(), data.id, data.category, data.name, data.date,
    data.eventName, data.scope, data.eventType, data.result,
    data.status, driveUrl
  ]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true, driveUrl: driveUrl }))
    .setMimeType(ContentService.MimeType.JSON);
}
